// --- Helper function: ISO 8601 Timestamp ---
String getISO8601Time() {
  struct tm timeinfo;
  if(!getLocalTime(&timeinfo)){
    return "Waiting..."; 
  }
  char timeStringBuff[50];
  strftime(timeStringBuff, sizeof(timeStringBuff), "%Y-%m-%dT%H:%M:%S+08:00", &timeinfo);
  return String(timeStringBuff);
}

// --- Helper function: Smooth Current ---
float getSmoothedCurrent() {
  float sum = 0;
  for(int i = 0; i < 10; i++) {
    sum += ina219.getCurrent_mA();
    delay(5);
  }
  return sum / 10.0;
}

void handleData() {
  float power_mW = cloud_voltage * cloud_current;
  
  if (cloud_current >= peakLimit) { isOverloaded = true; }
  int isOverload = isOverloaded ? 1 : 0; 
  
  float fridge_kW = isFridgeOn ? (sig_Fridge * 0.1) : 0.0;
  float tv_kW     = isTvOn ? (sig_Tv * 0.1) : 0.0;
  
  float leftover_mA = cloud_current - (isFridgeOn ? sig_Fridge : 0.0) - (isTvOn ? sig_Tv : 0.0) - (isLightOn ? sig_Light : 0.0);
  float ac_kW = (isAircondOn && leftover_mA > 2.5) ? (leftover_mA * 0.1) : 0.0;

  String json = "{";
  json += "\"current\":" + String(cloud_current, 1) + ",";
  json += "\"power\":" + String(power_mW, 1) + ",";
  json += "\"overload\":" + String(isOverload) + ",";
  
  json += "\"aircond\":\"" + String(isAircondOn ? "ON" : "OFF") + "\",";
  json += "\"tv\":\"" + String(isTvOn ? "ON" : "OFF") + "\",";
  json += "\"fridge\":\"" + String(isFridgeOn ? "ON" : "OFF") + "\",";
  json += "\"light\":\"" + String(isLightOn ? "ON" : "OFF") + "\",";
  json += "\"mosfet\":\"" + String(isMosfetOn ? "ON" : "OFF") + "\",";
  
  json += "\"fridge_hrs\":\"" + String(fridge_v_hours, 0) + "\",";

  json += "\"ac_hrs\":\"" + String(ac_v_hours, 0) + "\",";
  json += "\"light_hrs\":\"" + String(light_v_hours, 0) + "\",";

  json += "\"fridge_kw\":\"" + String(fridge_kW, 2) + "\",";
  json += "\"fridge_fee\":\"" + String(accumulated_fridge_cost, 4) + "\",";
  json += "\"tv_kw\":\"" + String(tv_kW, 2) + "\",";
  json += "\"tv_fee\":\"" + String(accumulated_tv_cost, 4) + "\",";

}

// PARALLEL TASK: HTTPS CLOUD SYNC (RUNS ON CORE 0)
// =========================================================
void cloudSyncTask(void * parameter) {
  for(;;) {
    // Wait exactly 2 seconds between cloud updates
    vTaskDelay(2000 / portTICK_PERIOD_MS); 
    
    if (WiFi.status() == WL_CONNECTED) {
      WiFiClientSecure *client = new WiFiClientSecure;
      client->setInsecure(); 

      HTTPClient http;
      http.begin(*client, backendURL);
      http.addHeader("Content-Type", "application/json");

      float c_A = cloud_current / 1000.0;
      float p_W = cloud_voltage * c_A;

      String json = "{";
      json += "\"timestamp\":\"" + getISO8601Time() + "\",";
      json += "\"power\":" + String(p_W, 3) + ",";
      json += "\"voltage\":" + String(cloud_voltage, 2) + ",";
      json += "\"current\":" + String(c_A, 4);
      json += "}";

      int code = http.POST(json);
      Serial.print("[Core 0] Cloud Sync Status: ");
      Serial.println(code);

      http.end();
      delete client; 
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(MOSFET_PIN, OUTPUT);
  digitalWrite(MOSFET_PIN, HIGH);
  isMosfetOn = true;

  pinMode(LDR_PIN, INPUT); 

  strip.begin();
  setPanelColor(40, 40, 40);

  if (!ina219.begin()) {
    Serial.println("INA219 ERROR");
    while (1);
  }

  WiFi.begin(ssid, password);
  Serial.print("Connecting WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi Connected!");
  Serial.print("🌐 Local Web UI URL: http://");
  Serial.println(WiFi.localIP());

  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);

  server.on("/", handleRoot);
  server.on("/analysis", handleAnalysis);
  server.on("/data", handleData);
  server.on("/shutdownMosfet", handleShutdownMosfet);
  server.on("/restoreMosfet", handleRestoreMosfet);
  server.on("/resetBill", handleResetBill);

  server.begin();
  previousCurrent = getSmoothedCurrent();
  lastBillingTime = millis();

  // --- START THE DUAL-CORE CLOUD TASK ---
  xTaskCreatePinnedToCore(
    cloudSyncTask,   // Function to run
    "CloudTask",     // Name of task
    10000,           // Stack size (HTTPS needs memory)
    NULL,            // Parameters
    1,               // Priority
    &CloudTask,      // Task Handle
    0                // Pin to Core 0 (Main loop runs on Core 1)
  );
  Serial.println("Dual-Core Processing Activated.");
}

// =========================================================
// LOOP (RUNS ON CORE 1 - LIGHTNING FAST)
// =========================================================
void loop() {
  server.handleClient(); // Answers the tablet instantly now!

  int lightState = digitalRead(LDR_PIN);
  isLightOn = (lightState == LOW); // Switch to HIGH if logic is inverted

  current_mA = getSmoothedCurrent();
  float voltage_V = ina219.getBusVoltage_V();

  // Share latest sensor data with Core 0 safely
  cloud_current = current_mA;
  cloud_voltage = voltage_V;

  // --- Real-Time Billing ---
  if (millis() - lastBillingTime >= 1000) {
    lastBillingTime = millis();
    
    if (isFridgeOn) {
      fridge_v_hours += 1.0;
      accumulated_fridge_cost += ((sig_Fridge * 0.1) * 1.0 * TNB_RATE_RM);
    }
    if (isTvOn) {
      tv_v_hours += 1.0;
      accumulated_tv_cost += ((sig_Tv * 0.1) * 1.0 * TNB_RATE_RM);
    }
    if (isLightOn) {
      light_v_hours += 1.0;
      accumulated_light_cost += (light_kW * 1.0 * TNB_RATE_RM);
    }
    
    float l_mA = current_mA - (isFridgeOn ? sig_Fridge : 0.0) - (isTvOn ? sig_Tv : 0.0) - (isLightOn ? sig_Light : 0.0);
    if (isAircondOn && l_mA > 2.5) { 
      ac_v_hours += 1.0;
      accumulated_ac_cost += ((l_mA * 0.1) * 1.0 * TNB_RATE_RM);
    }
    total_accumulated_cost = accumulated_fridge_cost + accumulated_tv_cost + accumulated_ac_cost + accumulated_light_cost;
  }

  // --- NILM Calculation States ---
  float currentDifference = current_mA - previousCurrent;

  if (current_mA < 1.0) {
    isAircondOn = false;
    isTvOn = false;
    isFridgeOn = false;
  }

  if (currentDifference > noiseFloor) {
    if (abs(currentDifference - sig_Fridge) <= tolerance) isFridgeOn = true;
    else if (abs(currentDifference - sig_Tv) <= tolerance) isTvOn = true;
    previousCurrent = current_mA;
    delay(200); // Reduced delay to keep web server fast
  }
  else if (currentDifference < -noiseFloor) {
    float drop = abs(currentDifference);
    if (abs(drop - sig_Fridge) <= tolerance) isFridgeOn = false;
    else if (abs(drop - sig_Tv) <= tolerance) isTvOn = false;
    previousCurrent = current_mA;
    delay(200); 
  }
  else {
    previousCurrent = (current_mA * 0.05) + (previousCurrent * 0.95);
  }

  float expectedFixedLoad = 0.0;
  if (isFridgeOn) expectedFixedLoad += sig_Fridge;
  if (isTvOn) expectedFixedLoad += sig_Tv;
  if (isLightOn) expectedFixedLoad += sig_Light; 

  float leftoverVariableCurrent = current_mA - expectedFixedLoad;
  isAircondOn = (leftoverVariableCurrent > 2.5);

  // --- NeoPixel LED ---
  int activeCount = 0;
  if (isFridgeOn) activeCount++;
  if (isTvOn) activeCount++;
  if (isAircondOn) activeCount++;

  if (current_mA >= peakLimit) {
    if (millis() - lastBlinkTime > 200) {
      lastBlinkTime = millis();
      blinkState = !blinkState;
      setPanelColor(blinkState ? 255 : 0, 0, 0);
    }
  }
  else if (activeCount >= 2) setPanelColor(0, 0, 255);
  else if (activeCount == 1) setPanelColor(0, 255, 0);
  else setPanelColor(40, 40, 40);

  delay(10);
}
